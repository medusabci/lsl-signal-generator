// Crear un entorno virtual de Python con solamente los paquetes que requiere la aplicación
cd "path/to/lsl-signal-generator"
python -m venv .venv
.\.venv\Scripts\activate

python -m pip install --upgrade pip
pip install -r requirements.txt

// Instalar Inno Setup (una sola vez)
winget install JRSoftware.InnoSetup

// Cambiar la versión: editar src\version.py (único sitio)

// Compilar con PyInstaller y generar el instalador
.\build_release.ps1

// Resultado: release\signal_generator_setup_<versión>.exe
// (opciones: -SkipBuild reutiliza dist\ ; -SkipInstaller solo ejecuta PyInstaller)
